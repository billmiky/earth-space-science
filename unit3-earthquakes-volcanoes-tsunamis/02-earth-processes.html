chart = {
  const width = 620;
  const height = 620;
  const speed = 0.004;
  const sphere = {type: "Sphere"};
  const graticule = d3.geoGraticule10();
  const context = DOM.context2d(width, height);
  const projection = d3.geoOrthographic()
    .fitExtent([[4, 4], [width - 4, height - 4]], sphere);
  const path = d3.geoPath(projection, context);
  const land = topojson.feature(world, world.objects.land);

  // Convert a hex color to an rgba() string with given alpha
  const hexRgba = (hex, a) => {
    const r = parseInt(hex.slice(1, 3), 16);
    const g = parseInt(hex.slice(3, 5), 16);
    const b = parseInt(hex.slice(5, 7), 16);
    return `rgba(${r},${g},${b},${a})`;
  };

  // Color by magnitude
  const magColor = mag => {
    if (mag >= 9.0) return "#ff2200";
    if (mag >= 8.5) return "#ff5500";
    if (mag >= 8.0) return "#ff8800";
    if (mag >= 7.5) return "#ffbb00";
    return "#ffdd00";
  };

  while (true) {
    projection.rotate([speed * performance.now(), -15]);
    context.clearRect(0, 0, width, height);
    context.save();

    // Ocean — radial gradient for a lit-globe look
    context.beginPath();
    path(sphere);
    const oceanGrad = context.createRadialGradient(
      width * 0.42, height * 0.36, width * 0.04,
      width * 0.50, height * 0.50, width * 0.48
    );
    oceanGrad.addColorStop(0, "#4eb8e8");
    oceanGrad.addColorStop(1, "#0b3d6e");
    context.fillStyle = oceanGrad;
    context.fill();

    // Land
    context.beginPath();
    path(land);
    const landGrad = context.createRadialGradient(
      width * 0.42, height * 0.36, width * 0.04,
      width * 0.50, height * 0.50, width * 0.48
    );
    landGrad.addColorStop(0, "#72c47a");
    landGrad.addColorStop(1, "#2a6040");
    context.fillStyle = landGrad;
    context.fill();
    context.strokeStyle = "#1e4d31";
    context.lineWidth = 0.4;
    context.stroke();

    // Graticule
    context.beginPath();
    path(graticule);
    context.globalAlpha = 0.18;
    context.strokeStyle = "#ffffff";
    context.lineWidth = 0.5;
    context.stroke();
    context.globalAlpha = 1;

    // Earthquake markers
    for (const eq of majorEarthquakes) {
      const coords = projection([eq.lon, eq.lat]);
      if (coords !== null) {
        const [x, y] = coords;
        const size = Math.max(3, (eq.magnitude - 6) * 4.5);
        const color = magColor(eq.magnitude);

        // Outer glow
        const glow = context.createRadialGradient(x, y, 0, x, y, size * 2.8);
        glow.addColorStop(0, hexRgba(color, 0.55));
        glow.addColorStop(1, hexRgba(color, 0));
        context.beginPath();
        context.arc(x, y, size * 2.8, 0, 2 * Math.PI);
        context.fillStyle = glow;
        context.fill();

        // Core dot
        context.beginPath();
        context.arc(x, y, size, 0, 2 * Math.PI);
        context.globalAlpha = 0.92;
        context.fillStyle = color;
        context.fill();
        context.globalAlpha = 1;
        context.strokeStyle = "rgba(255,255,255,0.85)";
        context.lineWidth = 0.9;
        context.stroke();
      }
    }

    // Globe outline
    context.beginPath();
    path(sphere);
    context.lineWidth = 1.5;
    context.strokeStyle = "#0a2a45";
    context.stroke();

    context.restore();
    yield context.canvas;
  }
}
